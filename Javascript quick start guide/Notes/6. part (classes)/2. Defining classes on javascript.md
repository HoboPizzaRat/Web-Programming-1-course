```javascript
class Address {
	// we can use the contructor to construct a fucking class with predefined properties
	// the constructor is always run when you create a class
	constructor(firstName, lastName) {
		this.firstName = firstName	
		this.lastName = lastName

	}
}

  
// you can pass the constructor method values during initialization
let customer = new Address("Robert", "Oliver")

for (let detail in customer) {
	console.log(detail)
}
```
